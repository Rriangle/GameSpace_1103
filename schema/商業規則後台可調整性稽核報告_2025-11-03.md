public async Task<int> CalculateExpRequiredAsync(int currentLevel)
{
    var formulaConfig = await GetSystemSettingAsync<FormulaConfig>("Pet.LevelUp.FormulaConfig");
    var nextLevel = currentLevel + 1;

    // 查找適用的公式層級
    var tier = formulaConfig.Tiers.FirstOrDefault(t => nextLevel >= t.MinLevel && nextLevel <= t.MaxLevel);

    if (tier == null) return 0;

    return tier.Formula switch
    {
        "linear" => (int)(tier.Params.A * nextLevel + tier.Params.B),
        "quadratic" => (int)(tier.Params.A * nextLevel * nextLevel + tier.Params.B),
        "exponential" => (int)(tier.Params.Base * Math.Pow(tier.Params.Rate, nextLevel)),
        _ => 0
    };
}

public class PetDailyDecayBackgroundService : BackgroundService
{
    private readonly IServiceProvider _services;
    private readonly IAppClock _appClock;

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        while (!stoppingToken.IsCancellationRequested)
        {
            var now = _appClock.ToAppTime(_appClock.UtcNow);
            var nextMidnight = now.Date.AddDays(1);
            var delay = nextMidnight - now;

            await Task.Delay(delay, stoppingToken);

            using var scope = _services.CreateScope();
            var decayService = scope.ServiceProvider.GetRequiredService<IPetDailyDecayService>();
            await decayService.ApplyDailyDecayToAllPetsAsync();
        }
    }
}

    var hungerDecay = await GetSystemSettingAsync<int>("Pet.DailyDecay.HungerDecay");
    var moodDecay = await GetSystemSettingAsync<int>("Pet.DailyDecay.MoodDecay");
    var staminaDecay = await GetSystemSettingAsync<int>("Pet.DailyDecay.StaminaDecay");
    var cleanlinessDecay = await GetSystemSettingAsync<int>("Pet.DailyDecay.CleanlinessDecay");
    var healthDecay = await GetSystemSettingAsync<int>("Pet.DailyDecay.HealthDecay");

    var pets = await _context.Pets.Where(p => !p.IsDeleted).ToListAsync();

    foreach (var pet in pets)
    {
        pet.Hunger = Math.Max(0, pet.Hunger - hungerDecay);
        pet.Mood = Math.Max(0, pet.Mood - moodDecay);
        pet.Stamina = Math.Max(0, pet.Stamina - staminaDecay);
        pet.Cleanliness = Math.Max(0, pet.Cleanliness - cleanlinessDecay);
        pet.Health = Math.Max(0, pet.Health - healthDecay);
    }

    await _context.SaveChangesAsync();
}
